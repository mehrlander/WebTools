The dock toggle appears only where the host installs window.__deckPane; this page installs one and qualifies it at 1024px and up, so the toggle is absent on a narrow window.

The docked edge is fixed here: dragging it needs kits/dock-split.js and a window.__deckWidth hook, and this page installs neither.

From a script

const inline = swipeDeck.core(notes.length, render, { innerClass: 'mx-auto' });
host.append(inline.track);           // a track with chrome of your own
inline.onSlide(i => say(i));

const deck = swipeDeck.open({        // the takeover
  count: notes.length, render, title: 'Field notes', icon: 'ph-notebook',
  index: i => ({ title: notes[i].title, group: notes[i].group }),
  actions: [{ icon: 'ph-shuffle', title: 'Random', onClick: d => d.go(pick()) }],
});
deck.setSubheader(strip);            // chrome belonging to the slide
swipeDeck.drill(deck, { count: 4, render: readings });   // one level down
deck.close();
