From a script

    
const d = Dictate.create({ onText: (t) => paint(t) });
d.start()               // needs a recognizer and microphone permission
d.punct(',')            // a real mark; the engine's own arrive as words
d.backWord()            // the last word, or the mark clinging to it
d.paragraphs(1500)      // the buffer with breaks proposed at the long pauses
d.flush()               // commit the running hypothesis, return the buffer
