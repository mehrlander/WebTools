From a script

const split = dockSplit.attach(handle, {
  axis: 'col', from: 'end', step: 2,
  bounds: () => panes.getBoundingClientRect(),   // what the percentage is of
  value: () => state.col,                        // where it is now
  min: () => 240 / panes.offsetWidth * 100,      // a pixel floor, read on every clamp
  onChange: pct => pane.style.width = pct + '%',
  onCommit: pct => localStorage.setItem('col', pct),
});
split.refresh();   // repaint the readout after the layout settles
split.destroy();   // listeners, readout and classes off

Capturing a drag

// drag-seam.mjs: the readout and the grown grip only exist mid-drag
export default async (page) => {
  const box = await page.locator('#col-handle').boundingBox();
  await page.mouse.move(box.x + 3, box.y + 140);
  await page.mouse.down();
  await page.mouse.move(box.x - 120, box.y + 140, { steps: 12 });
};
